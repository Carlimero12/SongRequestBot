                int MaxIterations = 0;
                int IterationNumber = 1;
                if ((e.ChatMessage.UserDetail.IsVip || Program.authUsers.Contains(e.ChatMessage.Username.ToLower())) && e.ChatMessage.Message.ToLower().Contains("batch"))
                {
                    MaxIterations = youtubeMatchID.Groups.Count;
                }
                while (IterationNumber < MaxIterations)
                {
                    IterationNumber++;
                    int IdListLenght = VidIdList.Count;
                    if (IdListLenght >= TLIBSecrets.config().maxSongs && TLIBSecrets.config().maxSongs != -1)
                    {
                        await YTPlaylistManagment.RemoveFromPlaylist();
                    }
                    string name = await YTPlaylistManagment.AddToPlaylist(youtubeMatchID.Groups[IterationNumber].Value);
                    Console.WriteLine(name);
                    if (!(MaxIterations > 0))
                    {
                        await client.SendMessageAsync(e.ChatMessage.Channel, $"Added {name} queue!");
                    }
                    if (IterationNumber > MaxIterations)
                    {break;}
                }
                if (MaxIterations > 0)
                {
                    await client.SendMessageAsync(e.ChatMessage.Channel, $"Added batch to the queue!");
